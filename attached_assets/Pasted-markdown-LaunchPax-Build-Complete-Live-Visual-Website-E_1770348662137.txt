markdown# LaunchPax: Build Complete Live Visual Website Editor System

## CONTEXT
You are building a critical platform feature for LaunchPax - an AI business operations platform. Currently, users can generate websites but editing is disconnected from the visual preview. You need to build a professional, live visual editor where users can click ANY element on their website and edit it instantly with real-time preview updates.

This is a PLATFORM-LEVEL feature that must work for ALL projects, ALL industries, ALL section types, and ALL content - not just one specific website.

## YOUR MISSION
Build a complete live visual editing system with split-screen layout: live website preview on the left, context-aware editor panel on the right. Users click any element (text, image, button, background, spacing, etc.) and the editor panel instantly shows relevant controls. Changes update in real-time without save buttons.

```tsx
// client/src/components/editor/VisualEditor.tsx

import { useState, useEffect } from 'react';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { InteractivePreview } from './InteractivePreview';
import { EditorPanel } from './EditorPanel';
import { SectionToolbar } from './SectionToolbar';
import { TopToolbar } from './TopToolbar';

export function VisualEditor({ projectId }: { projectId: string }) {
  const queryClient = useQueryClient();
  const [selectedElement, setSelectedElement] = useState(null);
  const [hoveredSection, setHoveredSection] = useState(null);
  const [editHistory, setEditHistory] = useState({ past: [], future: [] });

  // Fetch website data
  const { data: website } = useQuery({
    queryKey: ['website', projectId],
    queryFn: () => fetch(`/api/projects/${projectId}/website`).then(r => r.json())
  });

  // Real-time update mutation (optimistic updates)
  const updateMutation = useMutation({
    mutationFn: async (update: ElementUpdate) => {
      const response = await fetch(`/api/projects/${projectId}/website/update`, {
        method: 'PATCH',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(update)
      });
      return response.json();
    },
    onMutate: async (update) => {
      // Optimistic update for instant UI feedback
      await queryClient.cancelQueries({ queryKey: ['website', projectId] });
      const previousWebsite = queryClient.getQueryData(['website', projectId]);
      
      queryClient.setQueryData(['website', projectId], (old: any) => {
        return applyUpdateToWebsite(old, update);
      });

      return { previousWebsite };
    },
    onError: (err, update, context) => {
      // Rollback on error
      queryClient.setQueryData(['website', projectId], context?.previousWebsite);
    },
    onSettled: () => {
      queryClient.invalidateQueries({ queryKey: ['website', projectId] });
    }
  });

  // Handle element selection
  const handleElementClick = (element: SelectedElement) => {
    setSelectedElement(element);
    // Track in edit history
    setEditHistory(prev => ({
      past: [...prev.past, website],
      future: []
    }));
  };

  // Handle element changes (real-time)
  const handleElementChange = (path: string, value: any) => {
    updateMutation.mutate({
      projectId,
      path,
      value,
      timestamp: Date.now()
    });
  };

  // Undo/Redo
  const handleUndo = () => {
    if (editHistory.past.length === 0) return;
    const previous = editHistory.past[editHistory.past.length - 1];
    setEditHistory({
      past: editHistory.past.slice(0, -1),
      future: [website, ...editHistory.future]
    });
    queryClient.setQueryData(['website', projectId], previous);
  };

  const handleRedo = () => {
    if (editHistory.future.length === 0) return;
    const next = editHistory.future[0];
    setEditHistory({
      past: [...editHistory.past, website],
      future: editHistory.future.slice(1)
    });
    queryClient.setQueryData(['website', projectId], next);
  };

  if (!website) return ;

  return (
    
      
      {/* Top Toolbar */}
       0}
        canRedo={editHistory.future.length > 0}
        onPreview={() => window.open(`/preview/${projectId}`, '_blank')}
        onPublish={() => handlePublish(projectId)}
      />

      {/* Left: Live Preview */}
      
        
      

      {/* Right: Editor Panel */}
      
        <EditorPanel
          selectedElement={selectedElement}
          website={website}
          onChange={handleElementChange}
          onClose={() => setSelectedElement(null)}
        />
      

    
  );
}

// Helper to apply updates to website data structure
function applyUpdateToWebsite(website: any, update: ElementUpdate) {
  const cloned = JSON.parse(JSON.stringify(website));
  const pathParts = update.path.split('.');
  
  let current = cloned;
  for (let i = 0; i < pathParts.length - 1; i++) {
    const part = pathParts[i];
    const arrayMatch = part.match(/(.+)\[(\d+)\]/);
    
    if (arrayMatch) {
      const [, arrayName, index] = arrayMatch;
      current = current[arrayName][parseInt(index)];
    } else {
      current = current[part];
    }
  }
  
  const finalKey = pathParts[pathParts.length - 1];
  current[finalKey] = update.value;
  
  return cloned;
}

interface SelectedElement {
  type: 'text' | 'image' | 'section' | 'block' | 'style' | 'spacing';
  sectionId: string;
  sectionIndex: number;
  path: string; // e.g., "sections[0].hero.headline"
  value: any;
  label: string; // Human-readable label
  elementRef?: HTMLElement;
}

interface ElementUpdate {
  projectId: string;
  path: string;
  value: any;
  timestamp: number;
}

interface EditHistory {
  past: any[];
  future: any[];
}
```

```tsx
// client/src/components/editor/editors/TextEditor.tsx

import { useState } from 'react';
import { Label } from '@/components/ui/label';
import { Textarea } from '@/components/ui/textarea';
import { Button } from '@/components/ui/button';
import { Bold, Italic, Link } from 'lucide-react';

export function TextEditor({ value, onChange }: { value: string; onChange: (value: string) => void }) {
  
  const [localValue, setLocalValue] = useState(value);

  // Update on blur or every 500ms
  const handleChange = (newValue: string) => {
    setLocalValue(newValue);
    // Debounce real-time updates
    clearTimeout((window as any).textEditorTimeout);
    (window as any).textEditorTimeout = setTimeout(() => {
      onChange(newValue);
    }, 500);
  };

  return (
    
      
      
        Text Content
        <Textarea
          value={localValue}
          onChange={(e) => handleChange(e.target.value)}
          className="mt-2 min-h-[120px]"
          placeholder="Enter text content..."
        />
      

      {/* Character count */}
      
        {localValue.length} characters
      

      {/* Quick Actions */}
      
        <Button
          size="sm"
          variant="outline"
          onClick={() => {
            const upper = localValue.toUpperCase();
            setLocalValue(upper);
            onChange(upper);
          }}
        >
          UPPERCASE
        
        
        <Button
          size="sm"
          variant="outline"
          onClick={() => {
            const title = localValue
              .toLowerCase()
              .split(' ')
              .map(word => word.charAt(0).toUpperCase() + word.slice(1))
              .join(' ');
            setLocalValue(title);
            onChange(title);
          }}
        >
          Title Case
        
      

    
  );
}
```

```typescript
router.patch('/:projectId/website/update', async (req, res) => {
  try {
    const { projectId } = req.params;
    const { path, value, timestamp } = req.body;

    // Get current website data
    const project = await db.query.projects.findFirst({
      where: eq(projects.id, projectId),
      with: { websiteContent: true }
    });

    if (!project) {
      return res.status(404).json({ error: 'Project not found' });
    }

    // Apply update to website content
    const updatedContent = applyUpdate(
      project.websiteContent,
      path,
      value
    );

    // Save to database
    await db.update(websiteContent)
      .set({
        content: updatedContent,
        updatedAt: new Date()
      })
      .where(eq(websiteContent.projectId, projectId));

    // Track change in version history
    await db.insert(websiteVersionHistory).values({
      projectId,
      path,

      newValue: value,
      timestamp: new Date(timestamp)
    });

// Helper functions
function applyUpdate(content: any, path: string, value: any) {
  const cloned = JSON.parse(JSON.stringify(content));
  const pathParts = path.split('.');
  
  let current = cloned;
  for (let i = 0; i < pathParts.length - 1; i++) {
    const part = pathParts[i];
    const arrayMatch = part.match(/(.+)\[(\d+)\]/);
    
    if (arrayMatch) {
      const [, arrayName, index] = arrayMatch;
      current = current[arrayName][parseInt(index)];
    } else {
      current = current[part];
    }
  }
  
  const finalKey = pathParts[pathParts.length - 1];
  current[finalKey] = value;
  
  return cloned;
}

function getValueAtPath(content: any, path: string) {
  const pathParts = path.split('.');
  let current = content;
  
  for (const part of pathParts) {
    const arrayMatch = part.match(/(.+)\[(\d+)\]/);
    if (arrayMatch) {
      const [, arrayName, index] = arrayMatch;
      current = current[arrayName][parseInt(index)];
    } else {
      current = current[part];
    }
  }
  
  return current;
}
```

```typescript
// client/src/components/editor/KeyboardShortcuts.tsx

import { useEffect } from 'react';

export function useEditorKeyboardShortcuts({
  onUndo,
  onRedo,
  onSave,
  onPreview
}: {
  onUndo: () => void;
  onRedo: () => void;
  onSave: () => void;
  onPreview: () => void;
}) {
  
  useEffect(() => {
    const handleKeyboard = (e: KeyboardEvent) => {
      
      // Undo: Cmd/Ctrl + Z
      if ((e.metaKey || e.ctrlKey) && e.key === 'z' && !e.shiftKey) {
        e.preventDefault();
        onUndo();
      }

      // Redo: Cmd/Ctrl + Shift + Z
      if ((e.metaKey || e.ctrlKey) && e.key === 'z' && e.shiftKey) {
        e.preventDefault();
        onRedo();
      }

      // Save: Cmd/Ctrl + S
      if ((e.metaKey || e.ctrlKey) && e.key === 's') {
        e.preventDefault();
        onSave();
      }

      // Preview: Cmd/Ctrl + P
      if ((e.metaKey || e.ctrlKey) && e.key === 'p') {
        e.preventDefault();
        onPreview();
      }

      // Escape: Close editor panel
      if (e.key === 'Escape') {
        document.dispatchEvent(new CustomEvent('close-editor-panel'));
      }
    };

    window.addEventListener('keydown', handleKeyboard);
    return () => window.removeEventListener('keydown', handleKeyboard);
  }, [onUndo, onRedo, onSave, onPreview]);
}
```
